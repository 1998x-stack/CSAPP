cmake_minimum_required(VERSION 3.20)
project(CSAPP_Study LANGUAGES C CXX)

# Curated opt-in build: hundreds of .cpp files are teaching placeholders.
# Never glob everything into CMake targets.
include(CTest)
option(CSAPP_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer in supported compilers" OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(csapp_project_options INTERFACE)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(csapp_project_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(CSAPP_ENABLE_UBSAN)
    target_compile_options(csapp_project_options INTERFACE -fsanitize=undefined -fno-sanitize-recover=all)
    target_link_options(csapp_project_options INTERFACE -fsanitize=undefined -fno-sanitize-recover=all)
  endif()
elseif(CSAPP_ENABLE_UBSAN)
  message(FATAL_ERROR "CSAPP_ENABLE_UBSAN requires GCC or Clang for both C and C++.")
endif()

# Native x86-64 assembler is available only on supported Linux hosts.
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _cpu)
set(CSAPP_ASM_SUPPORTED OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND _cpu MATCHES "^(x86_64|amd64)$")
  enable_language(ASM)
  set(CSAPP_ASM_SUPPORTED ON)
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/examples/manifest.json" _manifest)
string(JSON _schema GET "${_manifest}" schema_version)
if(NOT _schema EQUAL 1)
  message(FATAL_ERROR "Unsupported examples manifest schema: ${_schema}")
endif()
string(JSON _count LENGTH "${_manifest}" examples)
if(_count LESS 1)
  message(FATAL_ERROR "The curated example manifest must not be empty.")
endif()
math(EXPR _last "${_count} - 1")
foreach(_index RANGE 0 ${_last})
  string(JSON _id GET "${_manifest}" examples ${_index} id)
  string(JSON _expected GET "${_manifest}" examples ${_index} expected)
  string(JSON _source_count LENGTH "${_manifest}" examples ${_index} sources)
  string(JSON _platform ERROR_VARIABLE _platform_error GET "${_manifest}" examples ${_index} platform)
  if(_platform_error)
    set(_platform "any")
  endif()
  if(NOT _platform STREQUAL "any" AND NOT _platform STREQUAL "linux-x86_64")
    message(FATAL_ERROR "Unknown target platform for ${_id}: ${_platform}")
  endif()
  if(_platform STREQUAL "linux-x86_64" AND NOT CSAPP_ASM_SUPPORTED)
    message(STATUS "Skipping Linux x86-64 lab on this host: ${_id}")
    continue()
  endif()
  string(REGEX MATCH "^[a-z][a-z0-9_]*$" _valid_id "${_id}")
  if(NOT _valid_id STREQUAL _id)
    message(FATAL_ERROR "Invalid CMake target id: ${_id}")
  endif()
  if(_source_count LESS 1)
    message(FATAL_ERROR "No sources provided for target: ${_id}")
  endif()
  math(EXPR _last_source "${_source_count} - 1")
  set(_source_files)
  set(_chapter_label "")
  foreach(_source_index RANGE 0 ${_last_source})
    string(JSON _source GET "${_manifest}" examples ${_index} sources ${_source_index})
    string(SUBSTRING "${_source}" 0 2 _chapter_of_source)
    if(_chapter_label STREQUAL "")
      set(_chapter_label "${_chapter_of_source}")
    elseif(NOT _chapter_label STREQUAL _chapter_of_source)
      message(FATAL_ERROR "Target crosses chapter directories: ${_id}")
    endif()
    if(IS_ABSOLUTE "${_source}" OR _source MATCHES "(^|/)\\.\\.(/|$)"
        OR NOT _source MATCHES "^(0[1-9]|1[0-2])_[^/]+/"
        OR NOT _source MATCHES "\\.(c|cc|cpp|cxx|s|S)$")
      message(FATAL_ERROR "Unsafe or unsupported source in manifest: ${_source}")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${_source}" OR IS_SYMLINK "${CMAKE_CURRENT_SOURCE_DIR}/${_source}")
      message(FATAL_ERROR "Missing or symlinked example: ${_source}")
    endif()
    if(_source MATCHES "\\.(s|S)$" AND NOT CSAPP_ASM_SUPPORTED)
      message(FATAL_ERROR "x86-64 assembly example requires Linux x86-64: ${_source}")
    endif()
    list(APPEND _source_files "${CMAKE_CURRENT_SOURCE_DIR}/${_source}")
  endforeach()
  add_executable(${_id} ${_source_files})
  target_link_libraries(${_id} PRIVATE csapp_project_options)
  if(BUILD_TESTING)
    add_test(NAME "example_${_id}" COMMAND ${_id})
    set_tests_properties("example_${_id}" PROPERTIES
      PASS_REGULAR_EXPRESSION "${_expected}" LABELS "example;chapter${_chapter_label}")
  endif()
endforeach()

# Compiler smoke tests are infrastructure fixtures, not completed CSAPP labs.
add_executable(csapp_c_smoke tests/fixtures/c_smoke.c)
target_link_libraries(csapp_c_smoke PRIVATE csapp_project_options)
if(BUILD_TESTING)
  add_test(NAME compiler_c_smoke COMMAND csapp_c_smoke)
  set_tests_properties(compiler_c_smoke PROPERTIES
    PASS_REGULAR_EXPRESSION "M03 C smoke: ok" LABELS "infrastructure;c")
endif()

# AT&T x86-64 assembly uses the System V ABI; optional on other hosts.
if(CSAPP_ASM_SUPPORTED)
  add_executable(csapp_asm_smoke
    tests/fixtures/asm_driver.c tests/fixtures/double_x86_64.S)
  target_link_libraries(csapp_asm_smoke PRIVATE csapp_project_options)
  if(BUILD_TESTING)
    add_test(NAME compiler_x86_64_asm_smoke COMMAND csapp_asm_smoke)
    set_tests_properties(compiler_x86_64_asm_smoke PROPERTIES
      PASS_REGULAR_EXPRESSION "M03 ASM smoke: ok" LABELS "infrastructure;asm")
  endif()
else()
  message(STATUS "x86-64 Linux assembly smoke test skipped on ${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}")
endif()
